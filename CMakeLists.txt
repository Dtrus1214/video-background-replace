cmake_minimum_required(VERSION 3.20)
project(VideoBackgroundReplace VERSION 0.1.0 LANGUAGES CXX)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
find_package(ONNXRuntime REQUIRED)
find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc imgcodecs videoio)
add_executable(video-background-replace src/main.cpp src/matanyone_onnx.cpp)
target_compile_features(video-background-replace PRIVATE cxx_std_17)
target_include_directories(video-background-replace PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(video-background-replace PRIVATE ${OpenCV_LIBS} ONNXRuntime::ONNXRuntime)
if(WIN32 AND ONNXRuntime_DLL)
  add_custom_command(TARGET video-background-replace POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${ONNXRuntime_DLL}" "$<TARGET_FILE_DIR:video-background-replace>")
endif()
if(WIN32)
  foreach(opencv_target IN LISTS OpenCV_LIBS)
    if(TARGET ${opencv_target})
      get_target_property(opencv_type ${opencv_target} TYPE)
      if(opencv_type STREQUAL "SHARED_LIBRARY")
        add_custom_command(TARGET video-background-replace POST_BUILD
          COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:${opencv_target}>" "$<TARGET_FILE_DIR:video-background-replace>")
      endif()
    endif()
  endforeach()
endif()
if(MSVC)
  target_compile_options(video-background-replace PRIVATE /W4 /permissive-)
else()
  target_compile_options(video-background-replace PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  add_test(NAME composite COMMAND video-background-replace --self-test)
endif()
